loops.f90
! loops.f90 -- counted loops, do while, exit and cycle
program loops
  implicit none
  integer, parameter :: wp = kind(1.0d0)
  integer  :: i, it, nt
  real(wp) :: total, time, dt, fuel

  ! 1) counted loop: sum of 1/i^2 tends to pi^2/6
  total = 0.0_wp
  do i = 1, 100000
    total = total + 1.0_wp / real(i, wp)**2
  end do
  print '(a, f12.8, a, f12.8)', 'sum 1/i^2 = ', total, '   pi^2/6 = ', acos(-1.0_wp)**2 / 6.0_wp

  ! 2) a step other than one, and counting down
  do i = 10, 1, -3
    write(*, '(i4)', advance='no') i
  end do
  print *

  ! 3) do while: the shape of FIRETEC's main time loop in compress.F90
  nt = 5; it = 0; time = 0.0_wp; dt = 0.01_wp
  do while (it < nt)
    it = it + 1
    time = time + dt
    print '(a, i3, a, f6.3)', 'step', it, '   time (s) =', time
  end do

  ! 4) exit leaves the loop, cycle skips to the next pass
  fuel = 1.0_wp
  do i = 1, 1000
    if (mod(i, 2) == 0) cycle          ! burn only on odd steps
    fuel = fuel * 0.8_wp
    if (fuel < 0.01_wp) exit           ! stop when less than 1% is left
  end do
  print '(a, i4, a, f7.4)', 'stopped at i =', i, ', fuel left =', fuel
end program loops

Download loops.f90

A counted loop is do i = start, stop [, step]. The loop variable must be an integer and must not be changed inside the loop. After an exit it keeps the value it had. FIRETEC's main program is a do while over time steps; every physics routine inside it is a set of nested counted loops over the grid.

Nested loops and memory order

Fortran stores a 2-D array column by column: a(1,1), a(2,1), a(3,1), …, a(1,2), … (C and Python's NumPy do the opposite). Memory is fastest when read in order, so the first index belongs in the innermost loop. That is why FIRETEC loops k (height) outside, j in the middle and i inside:

FIRETEC-1.5.1, Vegetation/burn_veg.F90 (excerpt)
  do k=1,lls
    do j=jl,ju
      do i=il,iu
        do ift=1,nfuel
          ! ... reaction rate for fuel type ift in cell (i,j,k)
loop_order.f90
! loop_order.f90 -- Fortran stores arrays column by column: loop over the FIRST index innermost
program loop_order
  implicit none
  integer, parameter :: wp = kind(1.0d0), n = 4000
  real(wp), allocatable :: a(:,:)
  integer :: i, j, c0, c1, rate
  real(wp) :: s

  allocate(a(n, n)); a = 1.0_wp

  call system_clock(c0, rate)
  s = 0.0_wp
  do j = 1, n            ! columns outside
    do i = 1, n          ! rows inside: walks through memory in order
      s = s + a(i, j)
    end do
  end do
  call system_clock(c1)
  print '(a, f8.4, a, f14.1)', 'i inner (fast): ', real(c1 - c0, wp) / rate, ' s, sum =', s

  call system_clock(c0)
  s = 0.0_wp
  do i = 1, n            ! rows outside
    do j = 1, n          ! columns inside: jumps n*8 bytes every step
      s = s + a(i, j)
    end do
  end do
  call system_clock(c1)
  print '(a, f8.4, a, f14.1)', 'j inner (slow): ', real(c1 - c0, wp) / rate, ' s, sum =', s
end program loop_order

Download loop_order.f90

With -O2 the compiler may reorder the slow loop for you, so try -O0 as well; on most machines the wrong order is several times slower. (In FIRETEC the innermost loop over fuel types runs over the first index of xvfuel(ift,i,j,k,var), also in order.)
