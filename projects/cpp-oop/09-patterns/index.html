Design patterns with abstract classes

Design patterns are named solutions to recurring design problems. Four of the most useful are built directly from abstract classes and virtual functions.

Strategy: interchangeable algorithms

A strategy is an algorithm behind an abstract interface, chosen at run time. In scientific code this separates what is solved from how: an abstract ODE describes dy/dt = f(t, y), an abstract Stepper describes one time step, and solve works with any combination. A new method or a new equation is a new class; solve never changes (the open/closed principle: open for extension, closed for modification).

integrators.cppRun online ↗
// integrators.cpp -- abstract classes in numerical work: one solver loop, interchangeable methods
#include <cmath>
#include <cstdio>
#include <memory>
#include <vector>

class ODE {                                         // dy/dt = f(t, y)
public:
    virtual ~ODE() = default;
    virtual double f(double t, double y) const = 0;
    virtual double exact(double t) const = 0;       // for measuring the error
};

class Decay : public ODE {                          // y' = -k y,  y(0) = 1
public:
    explicit Decay(double k) : k_(k) {}
    double f(double, double y) const override { return -k_ * y; }
    double exact(double t) const override { return std::exp(-k_ * t); }
private:
    double k_;
};

class Stepper {                                     // one step of a time-integration method
public:
    virtual ~Stepper() = default;
    virtual double step(const ODE& ode, double t, double y, double h) const = 0;
    virtual const char* name() const = 0;
};

class Euler : public Stepper {
public:
    double step(const ODE& ode, double t, double y, double h) const override { return y + h * ode.f(t, y); }
    const char* name() const override { return "Euler"; }
};

class Midpoint : public Stepper {
public:
    double step(const ODE& ode, double t, double y, double h) const override {
        double k1 = ode.f(t, y);
        return y + h * ode.f(t + h / 2, y + h / 2 * k1);
    }
    const char* name() const override { return "Midpoint (RK2)"; }
};

class RK4 : public Stepper {
public:
    double step(const ODE& ode, double t, double y, double h) const override {
        double k1 = ode.f(t, y);
        double k2 = ode.f(t + h / 2, y + h / 2 * k1);
        double k3 = ode.f(t + h / 2, y + h / 2 * k2);
        double k4 = ode.f(t + h, y + h * k3);
        return y + h / 6 * (k1 + 2 * k2 + 2 * k3 + k4);
    }
    const char* name() const override { return "Runge-Kutta 4"; }
};

// The solver knows nothing about which method or which equation it runs.
double solve(const ODE& ode, const Stepper& method, double y0, double T, int n) {
    double h = T / n, y = y0;
    for (int i = 0; i < n; ++i) y = method.step(ode, i * h, y, h);
    return y;
}

int main() {
    Decay ode(1.0);
    std::vector<std::unique_ptr<Stepper>> methods;
    methods.push_back(std::make_unique<Euler>());
    methods.push_back(std::make_unique<Midpoint>());
    methods.push_back(std::make_unique<RK4>());

    std::printf("%-16s %12s %12s %8s\n", "method", "err (n=10)", "err (n=20)", "ratio");
    for (const auto& m : methods) {
        double e10 = std::fabs(solve(ode, *m, 1.0, 1.0, 10) - ode.exact(1.0));
        double e20 = std::fabs(solve(ode, *m, 1.0, 1.0, 20) - ode.exact(1.0));
        std::printf("%-16s %12.3e %12.3e %8.1f\n", m->name(), e10, e20, e10 / e20);
    }
}
Output
method             err (n=10)   err (n=20)    ratio
Euler               1.920e-02    9.394e-03      2.0
Midpoint (RK2)      6.615e-04    1.592e-04      4.2
Runge-Kutta 4       3.332e-07    1.998e-08     16.7

Template Method and the non-virtual interface

The base class fixes the steps of an algorithm in a public, non-virtual function and lets derived classes fill in the steps through private virtual functions. The base keeps control (logging, checks, order of steps) and derived classes cannot skip it. This non-virtual interface (NVI) style is common in large C++ code bases.

nvi.cppRun online ↗
// nvi.cpp -- Template Method / Non-Virtual Interface: the base fixes the steps, derived classes fill them in
#include <cmath>
#include <iostream>
#include <string>

class Experiment {
public:
    virtual ~Experiment() = default;
    void run() {                                   // public and NOT virtual: the algorithm cannot be changed
        std::cout << "== " << name() << '\n';
        setup();
        double result = measure();                 // the steps that vary are private virtual functions
        std::cout << "   result " << result << (check(result) ? "  (ok)" : "  (out of range!)") << '\n';
    }
private:
    virtual std::string name() const = 0;
    virtual void setup() {}                        // optional step: a default that may be overridden
    virtual double measure() = 0;
    virtual bool check(double r) const { return std::isfinite(r); }
};

class Pendulum : public Experiment {
    double length_ = 0;
    std::string name() const override { return "pendulum period"; }
    void setup() override { length_ = 2.0; std::cout << "   length set to " << length_ << " m\n"; }
    double measure() override { return 2 * 3.14159 * std::sqrt(length_ / 9.81); }
    bool check(double r) const override { return r > 0 && r < 10; }
};

class FreeFall : public Experiment {
    std::string name() const override { return "free fall from 1 m"; }
    double measure() override { return std::sqrt(2 * 1.0 / 9.81); }
};

int main() {
    Pendulum p;
    FreeFall f;
    for (Experiment* e : {static_cast<Experiment*>(&p), static_cast<Experiment*>(&f)}) e->run();
}
Output

   result 2.837  (ok)

A derived class can override a private virtual function: access control decides who may call it, not who may override it.

Observer: notifying whoever is interested

A subject keeps a list of observers behind an abstract interface and notifies them of events. The subject never learns their concrete types, so new kinds of observers (a logger, a plot, a network sender) can be added without touching it.

observer.cppRun online ↗
// observer.cpp -- Observer: objects subscribe to events through an abstract interface
#include <iostream>
#include <string>
#include <vector>

class Observer {                                   // anything that wants to hear about new temperatures
public:
    virtual ~Observer() = default;
    virtual void update(double temperature) = 0;
};

class Sensor {                                     // the subject: knows only the interface, not the observers' types
public:
    void subscribe(Observer* o) { observers_.push_back(o); }
    void measure(double t) {
        std::cout << "sensor reads " << t << '\n';
        for (Observer* o : observers_) o->update(t);
    }
private:
    std::vector<Observer*> observers_;             // non-owning: observers must outlive the sensor (or unsubscribe)
};

class Display : public Observer {
public:
    void update(double t) override { std::cout << "  display: " << t << " C\n"; }
};

class Alarm : public Observer {
public:
    explicit Alarm(double limit) : limit_(limit) {}
    void update(double t) override { if (t > limit_) std::cout << "  ALARM: above " << limit_ << " C\n"; }
private:
    double limit_;
};

class Average : public Observer {
public:
    void update(double t) override { sum_ += t; ++n_; std::cout << "  running mean: " << sum_ / n_ << '\n'; }
private:
    double sum_ = 0;
    int n_ = 0;
};

int main() {
    Sensor s;
    Display d; Alarm a(30); Average m;
    s.subscribe(&d); s.subscribe(&a); s.subscribe(&m);
    for (double t : {21.5, 28.0, 33.5}) s.measure(t);
}

  running mean: 21.5
sensor reads 28

  running mean: 24.75
sensor reads 33.5

Lifetime The sensor stores plain pointers, so every observer must outlive it or unsubscribe first. Real code often stores std::weak_ptrs, or returns a subscription object whose destructor unsubscribes (RAII again).

Visitor: new operations without editing the classes

Lesson 08 noted that with inheritance, adding an operation means editing every class. The Visitor pattern turns that around: each class gets one accept function, and every new operation becomes a new visitor class. It works by double dispatch: s->accept(v) dispatches on the shape's type, and inside it v.visit(*this) dispatches on the visitor's type, with overloading picking the right visit.

visitor.cppRun online ↗
// visitor.cpp -- Visitor: add new operations to a fixed class hierarchy without editing it (double dispatch)
#include <cmath>
#include <iostream>
#include <memory>
#include <vector>

class Circle; class Rect;

class ShapeVisitor {                               // one visit() per concrete class
public:
    virtual ~ShapeVisitor() = default;
    virtual void visit(const Circle&) = 0;
    virtual void visit(const Rect&) = 0;
};

class Shape {
public:
    virtual ~Shape() = default;
    virtual void accept(ShapeVisitor& v) const = 0;   // first dispatch: on the shape's type
};

class Circle : public Shape {
public:
    explicit Circle(double r) : r(r) {}
    void accept(ShapeVisitor& v) const override { v.visit(*this); }   // second dispatch: overload picks visit(const Circle&)
    double r;
};

class Rect : public Shape {
public:
    Rect(double w, double h) : w(w), h(h) {}
    void accept(ShapeVisitor& v) const override { v.visit(*this); }
    double w, h;
};

// Two operations, written later, without touching Circle or Rect:
class Perimeter : public ShapeVisitor {
public:
    double total = 0;
    void visit(const Circle& c) override { total += 2 * 3.14159 * c.r; }
    void visit(const Rect& r) override { total += 2 * (r.w + r.h); }
};

class SvgWriter : public ShapeVisitor {
public:
    void visit(const Circle& c) override { std::cout << "<circle r=\"" << c.r << "\"/>\n"; }
    void visit(const Rect& r) override { std::cout << "<rect width=\"" << r.w << "\" height=\"" << r.h << "\"/>\n"; }
};

int main() {
    std::vector<std::unique_ptr<Shape>> shapes;
    shapes.push_back(std::make_unique<Circle>(1));
    shapes.push_back(std::make_unique<Rect>(2, 3));

    Perimeter p;
    for (const auto& s : shapes) s->accept(p);
    std::cout << "total perimeter " << p.total << '\n';

    SvgWriter svg;
    for (const auto& s : shapes) s->accept(svg);
}
Output
total perimeter 16.2832
<circle r="1"/>
<rect width="2" height="3"/>

The price is the reverse trade-off: adding a new shape now means adding a visit to every visitor. For a closed set of types, std::variant with std::visit does the same job with less code.

Key points

  • Strategy: one interface, interchangeable algorithms, chosen at run time.
  • Template Method / NVI: the base owns the sequence; derived classes supply the steps.
  • Observer decouples senders from receivers; Visitor adds operations to a fixed hierarchy.
