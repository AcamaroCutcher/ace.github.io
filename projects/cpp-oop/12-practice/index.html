1 · An ellipse

Add an Ellipse to the abstract Shape hierarchy of lesson 06. Its perimeter has no closed form: use Ramanujan's approximation π(a + b)(1 + 3h / (10 + √(4 − 3h))) with h = (a − b)² / (a + b)². Then make Circle derive from Ellipse. Does describe() need any change? Is the circle-is-an-ellipse relationship safe here, and would it be if shapes had set_axes()?

Show a solution
ellipse.cppRun online ↗
// ellipse.cpp -- exercise 1: add an Ellipse to the abstract Shape hierarchy
#include <cmath>
#include <iostream>
#include <memory>
#include <string>
#include <vector>

constexpr double pi = 3.141592653589793;

class Shape {
public:
    virtual ~Shape() = default;
    virtual double area() const = 0;
    virtual double perimeter() const = 0;
    virtual std::string name() const = 0;
    void describe() const { std::cout << name() << ": area " << area() << ", perimeter " << perimeter() << '\n'; }
};

class Ellipse : public Shape {
public:
    Ellipse(double a, double b) : a_(a), b_(b) {}
    double area() const override { return pi * a_ * b_; }
    double perimeter() const override {            // Ramanujan's second approximation
        double h = (a_ - b_) * (a_ - b_) / ((a_ + b_) * (a_ + b_));
        return pi * (a_ + b_) * (1 + 3 * h / (10 + std::sqrt(4 - 3 * h)));
    }
    std::string name() const override { return "ellipse"; }
protected:
    double a_, b_;
};

class Circle : public Ellipse {                    // safe here: shapes cannot be changed after construction
public:
    explicit Circle(double r) : Ellipse(r, r) {}
    std::string name() const override { return "circle"; }
};

int main() {
    std::vector<std::unique_ptr<Shape>> shapes;
    shapes.push_back(std::make_unique<Ellipse>(2, 1));
    shapes.push_back(std::make_unique<Circle>(1));
    for (const auto& s : shapes) s->describe();    // describe() did not change at all
    std::cout << "circle check: 2*pi = " << 2 * pi << '\n';
}
Output

circle: area 3.14159, perimeter 6.28319
circle check: 2*pi = 6.28319

2 · Sensors

Write an abstract Sensor with name(), read() and unit(), two implementations, and a function that prints a table of readings from a std::vector<std::unique_ptr<Sensor>>. Should read() be const?

Show a solution
sensors.cppRun online ↗
// sensors.cpp -- exercise 2: an abstract Sensor interface and a table of readings
#include <cstdio>
#include <memory>
#include <string>
#include <vector>

class Sensor {
public:
    virtual ~Sensor() = default;
    virtual std::string name() const = 0;
    virtual double read() = 0;                     // not const: reading may change internal state
    virtual const char* unit() const = 0;
};

class Thermometer : public Sensor {
public:
    explicit Thermometer(std::string where) : where_(std::move(where)) {}
    std::string name() const override { return "temperature (" + where_ + ")"; }
    double read() override { t_ += 0.5; return t_; }   // a fake, slowly warming room
    const char* unit() const override { return "C"; }
private:
    std::string where_;
    double t_ = 20;
};

class WindVane : public Sensor {
public:
    std::string name() const override { return "wind speed"; }
    double read() override { static const double seq[] = {3.2, 4.8, 2.1}; return seq[i_++ % 3]; }
    const char* unit() const override { return "m/s"; }
private:
    int i_ = 0;
};

void table(std::vector<std::unique_ptr<Sensor>>& sensors, int rounds) {
    for (int r = 1; r <= rounds; ++r)
        for (auto& s : sensors) std::printf("round %d  %-22s %6.1f %s\n", r, s->name().c_str(), s->read(), s->unit());
}

int main() {
    std::vector<std::unique_ptr<Sensor>> sensors;
    sensors.push_back(std::make_unique<Thermometer>("lab"));
    sensors.push_back(std::make_unique<WindVane>());
    table(sensors, 2);
}
Output

round 1  wind speed                3.2 m/s
round 2  temperature (lab)        21.0 C
round 2  wind speed                4.8 m/s

3 · A new method and a new equation

Add a Heun stepper and a second ODE, the logistic equation y' = y(1 − y) with exact solution 1 / (1 + e−t), to the integrators of lesson 09 without changing solve(). Confirm the error ratios.

Show a solution
heun.cppRun online ↗
// heun.cpp -- exercise 3: a new Stepper (Heun) and a new ODE (logistic), without changing solve()
#include <cmath>
#include <cstdio>
#include <memory>
#include <vector>

class ODE {
public:
    virtual ~ODE() = default;
    virtual double f(double t, double y) const = 0;
    virtual double exact(double t) const = 0;
    virtual const char* name() const = 0;
};

class Decay : public ODE {
public:
    double f(double, double y) const override { return -y; }
    double exact(double t) const override { return std::exp(-t); }
    const char* name() const override { return "y' = -y"; }
};

class Logistic : public ODE {                      // y' = y(1 - y), y(0) = 1/2
public:
    double f(double, double y) const override { return y * (1 - y); }
    double exact(double t) const override { return 1 / (1 + std::exp(-t)); }
    const char* name() const override { return "y' = y(1-y)"; }
};

class Stepper {
public:
    virtual ~Stepper() = default;
    virtual double step(const ODE& ode, double t, double y, double h) const = 0;
    virtual const char* name() const = 0;
};

class Euler : public Stepper {
public:
    double step(const ODE& o, double t, double y, double h) const override { return y + h * o.f(t, y); }
    const char* name() const override { return "Euler"; }
};

class Heun : public Stepper {                      // predictor-corrector, second order
public:
    double step(const ODE& o, double t, double y, double h) const override {
        double k1 = o.f(t, y);
        double k2 = o.f(t + h, y + h * k1);
        return y + h / 2 * (k1 + k2);
    }
    const char* name() const override { return "Heun"; }
};

double solve(const ODE& ode, const Stepper& m, double y0, double T, int n) {
    double h = T / n, y = y0;
    for (int i = 0; i < n; ++i) y = m.step(ode, i * h, y, h);
    return y;
}

int main() {
    Decay d; Logistic l;
    Euler e; Heun he;
    const ODE* odes[] = {&d, &l};
    const Stepper* methods[] = {&e, &he};
    std::printf("%-12s %-6s %11s %11s %6s\n", "equation", "method", "err n=10", "err n=20", "ratio");
    for (const ODE* o : odes) {
        double y0 = o->exact(0);
        for (const Stepper* m : methods) {
            double e10 = std::fabs(solve(*o, *m, y0, 1, 10) - o->exact(1));
            double e20 = std::fabs(solve(*o, *m, y0, 1, 20) - o->exact(1));
            std::printf("%-12s %-6s %11.3e %11.3e %6.1f\n", o->name(), m->name(), e10, e20, e10 / e20);
        }
    }
}
Output

y' = -y      Heun     6.615e-04   1.592e-04    4.2
y' = y(1-y)  Euler    2.344e-03   1.177e-03    2.0
y' = y(1-y)  Heun     1.038e-04   2.554e-05    4.1

4 · The rule of five

Write a class Grid that owns an n × n array through a raw double*, with a destructor, a deep copy constructor, a noexcept move constructor, and assignment. Hint: the copy-and-swap idiom gives both assignments in one function.

Show a solution
rule_of_five.cppRun online ↗
// rule_of_five.cpp -- exercise 4: a class that owns a raw array, with all five special members
#include <algorithm>
#include <cstddef>
#include <iostream>
#include <utility>

class Grid {                                       // n x n doubles on the heap
public:
    explicit Grid(std::size_t n) : n_(n), a_(new double[n * n]()) {}
    ~Grid() { delete[] a_; }
    Grid(const Grid& o) : n_(o.n_), a_(new double[o.n_ * o.n_]) {           // deep copy
        std::copy(o.a_, o.a_ + n_ * n_, a_);
    }
    Grid(Grid&& o) noexcept : n_(std::exchange(o.n_, 0)), a_(std::exchange(o.a_, nullptr)) {}   // steal
    Grid& operator=(Grid o) noexcept {             // copy-and-swap: one assignment covers copy and move
        swap(o);
        return *this;
    }
    void swap(Grid& o) noexcept { std::swap(n_, o.n_); std::swap(a_, o.a_); }
    double& operator()(std::size_t i, std::size_t j) { return a_[i * n_ + j]; }
    double operator()(std::size_t i, std::size_t j) const { return a_[i * n_ + j]; }
    std::size_t size() const { return n_; }
private:
    std::size_t n_;
    double* a_;
};

int main() {
    Grid g(3);
    g(1, 1) = 5;
    Grid copy = g;                // deep copy
    copy(1, 1) = 7;
    Grid moved = std::move(g);    // g is now empty
    Grid assigned(1);
    assigned = copy;              // copy-assign through copy-and-swap
    std::cout << "moved(1,1) = " << moved(1, 1) << ", copy(1,1) = " << copy(1, 1)
              << ", assigned(1,1) = " << assigned(1, 1) << ", g.size() = " << g.size() << '\n';
}
Output
moved(1,1) = 5, copy(1,1) = 7, assigned(1,1) = 7, g.size() = 0
